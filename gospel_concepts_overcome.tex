					\item[BDAG 596b] \textbf{}
						\begin{compactitem}
							\item[1] \textbf{to win in the face of obstacles, \textit{be victor, conquer, overcome, prevail}}, intransitive
							\item[1.a] in a battle or contest
							\item[1.b] in a legal action
							\item[2] \textbf{to overcome someone, \textit{vanquish, overcome}}, transitive
							\item[2.a] active with the object in the accusative \gr{τινά} \textit{overcome someone}...of Christ \gr{νενίκηκα τὸν κόσμον} \textit{I have overcome the world} (i.e. the sum total of everything opposed to God; see \hyperref[KOSMOS]{\grl{κόσμος} 7b}) (\hyperref[JOHN-16-33]{John 16:33})
							\item[2.b] passive \textit{be conquered, beaten}
							\item[3] \textbf{to surpass in ability, \textit{outstrip, excel}}, transitive with the superior power added in the dative
