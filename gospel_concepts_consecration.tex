\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{CONSECRATION-1828}
	
	% -----
	\subsubsection{Consecrate}
	% -----

		\begin{compactitem}
			\item[1] To make or declare to be sacred, by certain ceremonies or rites; to appropriate to sacred uses; to set apart, dedicate, or devote, to the service and worship of God.
			\item[2] To canonize; to exalt to the rank of a saint; to enroll among the gods, as a Roman emperor.
			\item[3] To set apart and bless the elements in the eucharist.
			\item[4] To render venerable; to make respected; as, rules or principles consecrated by time.
		\end{compactitem}
		
	% -----
	\subsubsection{Consecration}
	% -----
	
		\begin{compactitem}
			\item[1] The act or ceremony of separating from a common to a sacred use, or of devoting and dedicating a person or thing to the service and worship of God, by certain rites or solemnities.
			\item[2] Canonization; the act of translating into heaven, and enrolling or numbering among the saints or gods; the ceremony of the apotheosis of an emperor.
			\item[3] The benediction of the elements in the eucharist; the act of setting apart and blessing the elements in the communion.
		\end{compactitem}
