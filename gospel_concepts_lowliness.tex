	% -----
	\subsubsection{Lowliness}
	% -----

		\begin{compactitem}
			\item[1] Freedom from pride; humility; humbleness of mind.
			\item[2] Meanness; want of dignity; abject state. [In this sense \textit{Little Used.}]
		\end{compactitem}
		
	% -----
	\subsubsection{Lowly}
	% -----

		\begin{compactitem}
			\item[1] Having a low esteem of one's own worth; humble; meek; free from pride.
			\item[2] Mean; low; wanting dignity or rank.
			\item[3] Not lofty or sublime; humble.
			\item[4] Not high; not elevated in place.
		\end{compactitem}
