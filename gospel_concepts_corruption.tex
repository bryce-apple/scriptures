	% -----
	\subsubsection{Corruption}
	% -----

		\begin{compactitem}
			\item[1] The act of corrupting, or state of being corrupt or putrid; the destruction of the natural form of bodies, by the separation of the component parts, or by disorganization, in the process of putrefaction.
			\item[2] Putrid matter; pus.
			\item[3] Putrescence; a foul state occasioned by putrefaction.
			\item[4] Depravity; wickedness; perversion or deterioration of moral principles; loss of purity or integrity.
			\item[5] Debasement; taint; or tendency to a worse state.
			\item[6] Impurity; depravation; debasement; as a \textit{corruption} of language.
			\item[7] Bribery. He obtained his suit by \textit{corruption}
			\item[8] In law, taint; impurity of blood, in consequence of an act of attainder of treason or felony, by which a person is disabled to inherit lands from an ancestor, nor can retain those in his possession, nor transmit them by descent to his heirs.
		\end{compactitem}
		
	% -----
	\subsubsection{Corrupt (adjective)}
	% -----
	
		\begin{compactitem}
			\item[1] Changed from a sound to a putrid state, as by natural decomposition.
			\item[2] Spoiled; tainted; vitiated; unsound; as \textit{corrupt} air, or bread.
			\item[3] Depraved; vitiated; tainted with wickedness.
			\item[4] Debased; rendered impure; changed to a worse state; as \textit{corrupt} language.
			\item[5] Not genuine; infected with errors or mistakes. The text is \textit{corrupt}.
		\end{compactitem}
