		\paragraph{Greek New Testament} % *GREEK
		% --
			\label{SPRING-GREEK}
		
			%
			\subparagraph{\texorpdfstring{\gr{ἅλλομαι}}{}}
			%
				\label{HALLOMAI}
				
				For more discussion of this word, see \hyperref[JOHN-4-14]{John 4:14} and notes there.
			
				\begin{compactdesc}
					\item[BDAG 40b] \textbf{}
						\begin{compactitem}
							\item[1] to make a quick leaping movement, \textit{leap, spring up}
							\item[2] figurative, of the quick movement of inanimate things: to spring up from a source, of water \textit{well up, bubble up}
						\end{compactitem}
					\item[LSJ 70a, PDF 143] \textbf{}
						\begin{compactitem}
							\item[1] \textit{spring, leap} (properly of living beings)
							\item[2] with accusative, \textit{leap over}
							\item[3] of things; of sound; of parts of body, \textit{twitch, quiver, throb}
						\end{compactitem}
				\end{compactdesc}
				
				\begin{compactdesc}
					\item[Some NT uses] \textbf{}
						\begin{compactdesc}
							\item[John 4:14] \gr{ὃς δ᾽ ἂν πίῃ ἐκ τοῦ ὕδατος οὗ ἐγὼ δώσω αὐτῷ, οὐ μὴ διψήσει εἰς τὸν αἰῶνα, ἀλλὰ τὸ ὕδωρ ὃ δώσω αὐτῷ γενήσεται ἐν αὐτῷ πηγὴ ὕδατος ἁλλομένου εἰς ζωὴν αἰώνιον.}
							\item[Acts 3:8] \gr{καὶ ἐξαλλόμενος ἔστη καὶ περιεπάτει, καὶ εἰσῆλθεν σὺν αὐτοῖς εἰς τὸ ἱερὸν περιπατῶν καὶ ἁλλόμενος καὶ αἰνῶν τὸν θεόν.}
							\item[Acts 14:10] \gr{εἶπεν μεγάλῃ φωνῇ· Ἀνάστηθι ἐπὶ τοὺς πόδας σου ὀρθός· καὶ ἥλατο καὶ περιεπάτει.} \textit{(\gr{ἥλατο} is aorist middle indicative.)}
						\end{compactdesc}
				\end{compactdesc}
