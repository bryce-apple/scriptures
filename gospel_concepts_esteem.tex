\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{ESTEEM-1828}

	% -----
	\subsubsection{Verb}
	% -----

		\begin{compactitem}
			\item[1] To set a value on, whether high or low; to estimate; to value.
			\item[2] To prize; to set a high value on; to regard with reverence, respect or friendship.
			\item[3] To hold in opinion; to repute; to think.
			\item[4] To compare in value; to estimate by proportion. [\textit{Little Used.}]
		\end{compactitem}
		
	% -----
	\subsubsection{Noun}
	% -----
	
		\begin{compactitem}
			\item[1] High value or estimation; great regard; favorable opinion, founded on supposed worth.
		\end{compactitem}
