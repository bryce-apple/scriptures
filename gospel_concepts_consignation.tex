\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{CONSIGNATION-1828}
	
	% -----
	\subsubsection{Consignation}
	% -----
	
		\begin{compactitem}
			\item The act of consigning; the act of delivering or committing to another person, place or state.
		\end{compactitem}
	
	% -----
	\subsubsection{Consign}
	% -----

		\begin{compactitem}
			\item[1] To give, send or set over; to transfer or deliver into the possession of another, or into a different state, with the sense of fixedness in that state, or permanence of possession.
			\item[2] To deliver or transfer, as a charge or trust; to commit; as, to \textit{consign} a youth to the care of a preceptor; to \textit{consign} goods to a factor.
			\item[3] To set over or commit, for permanent preservation; as, to \textit{consign} a history to writing.
			\item[4] To appropriate.
		\end{compactitem}
