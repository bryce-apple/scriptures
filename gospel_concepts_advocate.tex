\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{ADVOCATE-1828}
	
	% -----
	\subsubsection{Noun}
	% -----

		\begin{compactitem}
			\item[1] \textit{advocate} in its primary sense, signifies, one who pleads the cause of another in a court of civil law. Hence,
			\item[2] One who pleads the cause of another before any tribunal or judicial court, as a barrister in the English courts. We say, a man is a learned lawyer and an able \textit{advocate}
			\item[3] One who defends, vindicates, or espouses a cause, by argument; one who is friendly to; as, an \textit{advocate} for peace, or for the oppressed.
		\end{compactitem}
		
	% -----
	\subsubsection{Verb}
	% -----

		\begin{compactitem}
			\item To plead in favor of; to defend by argument, before a tribunal; to support or vindicate.
		\end{compactitem}
