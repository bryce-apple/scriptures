\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{SELFEXISTENT-1828}

	\begin{compactitem}
		\item Existing by its own nature or essense, independent of any other cause. God is the only \textit{self-existent} being.
	\end{compactitem}

% ----------------------
\subsection{Oxford English Dictionary} % *OED
% ----------------------
	\label{SELFEXISTENT-OED}

	\begin{compactitem}
		\item[1] Having the property of self-existence; existing by virtue of intrinsic nature, independently of any other cause.
		\item[2] Having an independent existence.
	\end{compactitem}
