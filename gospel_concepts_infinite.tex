\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{INFINITE-1828}

	\begin{compactitem}
		\item[1] Without limits; unbounded; boundless; not circumscribed; applied to time, space and qualities.
		\item[2] That will have no end. Thus angels and men, though they have had a beginning, will exist in \textit{infinite} duration.
		\item[3] That has a beginning in space, but is infinitely extended; as, a line beginning at a point, but extended indefinitely, is an \textit{infinite} line.
		\item[4] \textit{infinite} is used loosely and hyperbolically for indefinitely large, immense, of great size or extent.
	\end{compactitem}
