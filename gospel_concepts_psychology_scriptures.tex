	\subsubsection{Book of Mormon} % *BOM
	% -----
		\label{PSYCHOLOGY-BOM}
	
		% \begin{tabular}{ll}
			% Total occurrences in the BOM & 
		% \end{tabular}
		
		\begin{compactdesc}
			\item[{\hyperref[1NEPHI-2-11]{1 Nephi 2:11}}]
			\item[{\hyperref[1NEPHI-15-11]{1 Nephi 15:11}}] \phantom{.}
				\begin{compactitem}
					\item I think this is another place, and this is starting to open my eyes a little more to this elsewhere, where ``heart'' should be understood as what we understand by ``mind,'' where we're speaking more of the intellectual part of us.
					\item So ``hardening your heart'' would have to do with your willingness to learn, or change your ideas, or something like that.
				\end{compactitem}
			\item[{\hyperref[1NEPHI-16-24]{1 Nephi 16:24}}] \phantom{.}
				\begin{compactitem}
					\item Nephi does something with all the ``energy'' of his soul, which is similar to what it says in \hyperref[MORONI-7-48]{Moroni 7:48}. Possibly take ``soul'' and ``heart'' as equivalent in certain ways?
				\end{compactitem}
			\item[{\hyperref[1NEPHI-17-45]{1 Nephi 17:45}}]
			\item[{\hyperref[MOSIAH-5-13]{Mosiah 5:13}}]
			\item[{\hyperref[MOSIAH-24-10]{Mosiah 24:10--12}}] \phantom{.}
				\begin{compactitem}
					\item In this passage, it seems clear that the ``heart'' is what we would call the ``mind,'' or the intellectual part of us; it talks about the ``thoughts of their hearts.''
					\item I'm guessing there are other passages like this, and I'll look for them eventually.
				\end{compactitem}
			\item[{\hyperref[ALMA-12-7]{Alma 12:7}}]
			\item[{\hyperref[ALMA-16-16]{Alma 16:16--17}}]
			\item[{\hyperref[ALMA-18-20]{Alma 18:20}}]
