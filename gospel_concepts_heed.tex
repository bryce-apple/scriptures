\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{HEED-1828}
	
	% -----
	\subsubsection{Transitive verb}
	% -----

		\begin{compactitem}
			\item To mind; to regard with care; to take notice of; to attend to; to observe.
		\end{compactitem}
		
	% -----
	\subsubsection{Noun}
	% -----
	
		\begin{compactitem}
			\item[1] Caution; care; watch for danger; notice; circumspection; usually preceded by take.
			\item[2] Notice; observation; regard; attention; often preceded by give.
			\item[3] Seriousness; a steady look.
		\end{compactitem}
