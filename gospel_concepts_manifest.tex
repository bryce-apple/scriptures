\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{MANIFEST-1828}
	
	% -----
	\subsubsection{Verb}
	% -----

		\begin{compactitem}
			\item[1] To reveal; to make to appear; to show plainly; to make public; to discple to the eye or to the understanding.
			\item[2] To display; to exhibit more clearly to the view.
		\end{compactitem}
	
	% -----
	\subsubsection{Adjective}
	% -----

		\begin{compactitem}
			\item[1] Plain, open, clearly visible to the eye or obvious to the understanding; apparent; not obscure or difficult to be seen or understood. From the testimony, the truth we conceive to be \textit{manifest}.
			\item[2] Detected; with \textit{of}.
		\end{compactitem}

% ----------------------
\subsection{Oxford English Dictionary} % *OED
% ----------------------
	\label{MANIFEST-OED}
	
	% -----
	\subsubsection{Verb}
	% -----

		\begin{compactitem}
			\item[1.a] To make (a quality, fact, etc.) evident to the eye or to the understanding; to show plainly, disclose, reveal.
			\item[1.b] Of a thing: to be evidence of, prove, attest.
			\item[1.c] With clause as object, or with object and infinitive or complement.
			\item[2.a] To display (a quality, condition, feeling, etc.) by action or behaviour; to give evidence of possessing, reveal the presence of, evince.
			\item[2.b] (\textit{reflexive} or in \textit{passive}). To be seen to be, or reveal itself as existing or operative.
			\item[3] To expound, unfold, clear up (a matter), as in a manifesto or public declaration. \textit{Obsolete}.
			\item[4] Chiefly \textit{Nautical}. To record or enumerate in a manifest. Frequently in \textit{passive}.
			\item[5] In Spanish law: to protect (a person) by a `manifestation'. \textit{Obsolete, rare}.
			\item[6] Originally \textit{Spiritualism}. Of a ghost or spirit: to reveal its presence, make an appearance.
			\item[7] = ``Demonstrate'' (verb, meaning \#7). \textit{(This is the sense of ``demonstrate'' that means to show out in support of cause, like ``The picketers were demonstrating in front of the building.'')}
		\end{compactitem}
