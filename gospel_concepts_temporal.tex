\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{TEMPORAL-1828}

	\begin{compactitem}
		\item[1] Pertaining to this life or this world or the body only; secular; as \textit{temporal} concerns; \textit{temporal} affairs. In this sense, it is opposed to spiritual. Let not \textit{temporal} affairs or employments divert the mind from spiritual concerns, which are far more important.
		\item[2] Measured or limited by time, or by this life or this state of things; having limited existence; opposed to eternal.
		\item[3] In grammar, relating to a tense; as a \textit{temporal} augment.
		\item[4] Pertaining to the temple or temples of the head; as the \textit{temporal} bone; a \textit{temporal} artery or vein; \textit{temporal} muscle.
	\end{compactitem}
