\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{OUTWARD-1828}

	\begin{compactitem}
		\item[1] External; exterior; forming the superficial part
		\item[2] External; visible
		\item[3] Extrinsic; adventitious
		\item[4] Foreign; not intestine
		\item[5] Tending to the exterior part
		\item[6] In Scripture, civil; public; as opposed to religious
		\item[7] In theology, carnal; fleshly; corporeal; not spiritual; as the \textit{outward} man
	\end{compactitem}
