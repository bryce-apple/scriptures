				\begin{compactdesc}
					\item[BDAG 502b] \textbf{}
						\begin{compactitem}
							\item[1] to make a selection, \textit{select, prefer}
							\item[2] to pass judgment upon (and thereby seek to influence) the lives and actions of other people
							\item[2a] \textit{judge, pass judgment upon, express an opinion about}
							\item[2b] especially \textit{pass an unfavorable judgment upon, criticize, find fault with, condemn}
							\item[3] to make a judgment based on taking various factors into account, \textit{judge, think, consider, look upon}
							\item[4] to come to a conclusion after a cognitive process, \textit{reach a decision, decide, propose, intend}
							\item[5] to engage in a judicial process, \textit{judge, decide, hale before a court, condemn}, also \textit{hand over for judicial punishment}
							\item[5a] of a human court
							\item[5a$\alpha$] active and passive absolute
							\item[5a$\beta$] middle and passive: ``dispute, quarrel, debate''
							\item[5b] of the divine tribunal
							\item[5b$\alpha$] occupied by God or Christ: absolute \textit{administer justice, judge}
							\item[5b$\beta$] occupied by those who have been divinely commissioned to judge
							\item[6] to ensure justice for someone, \textit{see to it that justice is done}
						\end{compactitem}
					\item[CGL 835a] \textbf{}
						\begin{compactitem}
							\item The sections of meanings are grouped as: (1--3) separate, distinguish, divide, (4--5) choose, (6--10) determine (especially the outcome of a dispute), (11--18) make a decision or judgment, (19--21) give a legal judgment, (22) interrogate.
							\item[1] \textbf{separate} (from each other)
							\item[2] distinguish (from each other)---
							\item[19] be judge, give judgment
							\item[20] bring to trial
							\item[21] pass judgment on, judge---someone \textit{(CGL lists the NT uses of the term under this meaning.)}
						\end{compactitem}

	\subsubsection{Key Points}
	% -----
		\label{JUDGE-KEYS}
	
		\begin{compactitem}
			\item In the scriptures, ``judgment'' can sometimes mean ``consequence of the law,'' as in \hyperref[2NEPHI-9-7]{2 Nephi 9:7}; it doesn't have to be a verdict given by a person or God. See also \hyperref[2NEPHI-9-25]{2 Nephi 9:25} and \hyperref[ALMA-42-17]{Alma 42:17--22}, especially 22: ``Otherwise justice claimeth the creature and executeth the law, and the law inflicteth the punishment'' (that is, the punishment occurs in accordance with the law).
				\begin{compactitem}
					\item You can see this meaning of ``judgment'' as a verdict given by someone in an indirect way, if that person set up or established the law; but this meaning of ``judgment'' calls attention to the consequence of the law, rather than the giving of the verdict by a judge.
				\end{compactitem}
			\item We are instructed to ``judge righteously''
				\begin{compactitem}
					\item \hyperref[DC11-12]{DC 11:12}; 
				\end{compactitem}
		\end{compactitem}
