\subsection{Key Phrases} % *KEYPHRASES *PHRASES
% ----------------------
	\label{WATER-PHRASES}

	\begin{compactdesc}
	
		\item[living waters/waters of life] \textbf{22} | Song of Solomon 4:15; Jeremiah 2:13, 17:13; Zechariah 14:8; John 4:10, 11, 14; 7:38; Revelation 7:17, 21:6, 22:1, 17; 1 Nephi 11:25; Alma 5:34; 42:27; DC 10:66; 63:23; 133:29
			\begin{compactdesc}
				\item[Bible anchor verses] \phantom{.}
					\begin{compactdesc}
						\item[Jeremiah 2:13] \he{m:qOr mayim .hay*iym}
						\item[Jeremiah 17:13] \he{m:qOr mayim--.hay*iym}
						\item[Zechariah 14:8] \he{mayim--.hay*iym}
						\item[John 4:10] \gr{ὕδωρ ζῶν}
						\item[John 4:11] \gr{τὸ ὕδωρ τὸ ζῶν}
						\item[John 4:14] \gr{πηγὴ ὕδατος ἁλλομένου εἰς ζωὴν αἰώνιον}
						\item[John 7:38] \gr{ὕδατος ζῶντος}
						\item[Revelation 7:17] \gr{ἐπὶ ζωῆς πηγὰς ὑδάτων} \textit{(Maybe, ``to fountains of the waters of life.'')}
						\item[Revelation 21:6] \gr{ἐκ τῆς πηγῆς τοῦ ὕδατος τῆς ζωῆς}
						\item[Revelation 22:1] \gr{ποταμὸν ὕδατος ζωῆς}
						\item[Revelation 22:17] \gr{λαβέτω ὕδωρ ζωῆς}
					\end{compactdesc}
				\item[Commentary] See above at \hyperref[HUDOR]{\grl{ὕδωρ}}. \hyperref[1NEPHI-11-25]{1 Nephi 11:25} says that the ``waters are a representation of the love of God.'' But I think it's also useful to think of the more broadly than that as well
			\end{compactdesc}
			
	\end{compactdesc}
