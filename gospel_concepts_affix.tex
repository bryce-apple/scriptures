\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{AFFIX-1828}

	\begin{compactitem}
		\item[1] To unite at the end; to subjoin, annex, or add at the close; as, to \textit{affix} a syllable to a word; to \textit{affix} a seal to an instrument.
		\item[2] To attach, unite, or connect with, as names \textit{affixed} to ideas, or ideas \textit{affixed} to things.
		\item[3] To fix or fasten in any manner. In this sense, fix is more generally used.
	\end{compactitem}
