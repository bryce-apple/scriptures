	\subsubsection{The importance of ``doing the will'' of God}
	% -----
		\label{WILL-doing}
		
		\begin{compactitem}
			\item \hyperref[MATTHEW-7-21]{Matthew 7:21--23}
				\begin{compactitem}
					\item Compare \hyperref[LUKE-13-25]{Luke 13:25--27}
				\end{compactitem}
			\item \hyperref[MATTHEW-7-24]{Matthew 7:24--27}
			\item \hyperref[LUKE-12-47]{Luke 12:47--48}
			\item \hyperref[JOHN-9-31]{John 9:31}
			\item \hyperref[COLOSSIANS-1-9]{Colossians 1:9--13}
				\begin{compactitem}
					\item These verses aren't so much about doing the will but do talk about knowing it, and then they mention specific consequences of knowing the will, including behavior.
				\end{compactitem} 
			\item \hyperref[HELAMAN-15-17]{Helaman 15:17}
				\begin{compactitem}
					\item Note how this is connected to the knowledge of the Nephites, and marks a difference between them and the Lamantes; see verses 4--17 of this chapter.
				\end{compactitem}
		\end{compactitem}
