\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{ENEMY-1828}

	\begin{compactitem}
		\item[1] A foe; an adversary. A private \textit{enemy} is one who hates another and wishes him injury, or attempts to do him injury to gratify his own malice or ill will. A public \textit{enemy} or foe, is one who belongs to a nation or party, at war with another.
		\item[2] One who hates or dislikes; as an \textit{enemy} to truth or falsehood.
		\item[3] In theology, and by way of eminence, the \textit{enemy} is the Devil; the archfiend.
		\item[4] In military affairs, the opposing army or naval force in war, is called the \textit{enemy}.
	\end{compactitem}

\subsection{Key Phrases} % *KEYPHRASES *PHRASES
% ----------------------
	\label{ENEMY-PHRASES}

	\begin{compactdesc}
		\item[enemy of all righteousness] \hyperref[ACTS-13-10]{Acts 13:10}; \hyperref[MOSIAH-2-37]{Mosiah 2:37}; \hyperref[MOSIAH-4-14]{Mosiah 4:14}; \hyperref[ALMA-34-23]{Alma 34:23}; \hyperref[MORONI-9-6]{Moroni 9:6}
			\begin{compactdesc}
				\item[Commentary] Note that, of the four BOM uses, only Moroni 9:6 talks about ``\textit{the} enemy of all righteousness.'' The other three refer to ``an'' enemy. I'm not sure that the Moroni 9:6 use refers to Satan or the devil, however; see \hyperref[MORONI-9-6-NOTE-enemy]{Moroni 9:6}.
			\end{compactdesc}
		\item[enemy to God] \hyperref[MOSIAH-2-38]{Mosiah 2:38}; \hyperref[MOSIAH-3-19]{Mosiah 3:19}; \hyperref[MOSIAH-16-15]{Mosiah 16:15} (2)
	\end{compactdesc}
