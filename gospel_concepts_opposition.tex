\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{OPPOSITION-1828}

	\begin{compactitem}
		\item[1] Situation so as to front something else; a standing over against; as the \textit{opposition} of two mountains or buildings.
		\item[2] The act of opposing; attempt to check, restrain or defeat.
		\item[3] Obstacle.
		\item[4] Resistance; as the \textit{opposition} of enemies.
		\item[5] Contrariety; repugnance in principle; as the \textit{opposition} of the heart to the laws of God.
		\item[6] Contrariety of interests, measures on designs.
		\item[7] Contrariety or diversity of meaning; as one term used in \textit{opposition} to another. 
		\item[8] Contradiction; inconsistency.
		\item[9] The collective body of opposers; in England, the party in Parliament which opposed the ministry; in America, the party that opposed the existing administration.
		\item[10] In astronomy, the situation of two heavenly bodies, when distant from each other 180 degrees.
	\end{compactitem}
