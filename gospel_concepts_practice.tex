	% -----
	\subsubsection{Practice (transitive verb)}
	% -----

	\begin{compactitem}
		\item[1] To do or perform frequently, customarily or habitually; to perform by a succession of acts; as, to \textit{practice} gaming; to \textit{practice} fraud or deception; to \textit{practice} the virtues of charity and beneficence; to \textit{practice} hypocrisy.
		\item[2] To use or exercise any profession or art; as, to \textit{practice} law or medicine; to \textit{practice} gunnery or surveying.
		\item[3] To use or exercise for instruction, discipline or dexterity. [In this sense, the verb is usually intransitive.]
		\item[4] To commit; to perpetrate; as the horrors \textit{practice}d at Wyoming.
		\item[5] To use; as a \textit{practice}d road. [Unusual.]
	\end{compactitem}
