	\subsubsection{Key Points}
	% -----
		\label{CREATION-KEYS}
	
		\begin{compactitem}
			\item My own view about the creation stories in the scriptures
				\begin{compactitem}
					\item I don't think they are literally true.
					\item I also don't think that they are constructed in such a way that they can support a heavy symbolic interpretation throughout.
				\end{compactitem}
		\end{compactitem}
