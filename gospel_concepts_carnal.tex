\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{CARNAL-1828}

	\begin{compactitem}
		\item[1] Pertaining to flesh; fleshly; sensual; opposed to spiritual; as \textit{carnal} pleasure.
		\item[2] Being in the natural state; unregenerate. The \textit{carnal} mind is enmity against God. \hyperref[ROMANS-8-7]{Romans 8:7}.
		\item[3] Pertaining to the ceremonial law; as \textit{carnal} ordinances. \hyperref[HEBREWS-9-10]{Hebrews 9:10}.
		\item[4] Lecherous; lustful; libidinous; given to sensual indulgence.
	\end{compactitem}
