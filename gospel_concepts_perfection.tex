	% -----
	\subsubsection{Perfect}
	% -----

		\begin{compactitem}
			\item[1] Finished; complete; consummate; not defective; having all that is requisite to its nature and kind; as a \textit{perfect} statue; a \textit{perfect} likeness; a \textit{perfect} work; a \textit{perfect} system. As full, as \textit{perfect} in a hair as heart.
			\item[2] Fully informed; completely skilled; as men \textit{perfect} in the use of arms; \textit{perfect} in discipline.
			\item[3] Complete in moral excellencies. Be ye therefore \textit{perfect} even as your Father who is in heaven is \textit{perfect} Matthew 5:48.
			\item[4] Manifesting \textit{perfect}ion. My strength is made \textit{perfect} in weakness. 2 Corinthians 12:9. \textit{perfect} chord, in music, a concord or union of sounds which is \textit{perfect}ly coalescent and agreeable to the ear, as the fifth and the octave; a \textit{perfect} consonance. A \textit{perfect} flower, in botany, has both stamen and pistil, or at least another and stigma.
		\end{compactitem}
		
	% -----
	\subsubsection{Perfection}
	% -----

		\begin{compactitem}
			\item The state of being perfect or complete, so that nothing requisite is wanting; as \textit{perfection} in an art or science; \textit{perfection} in a system of morals.
			\item[1] Physical \textit{perfection} is when a natural object has all its powers, faculties or qualities entire and in full vigor, and all its parts in due proportion.
			\item[2] Metaphysical or transcendental \textit{perfection} is the possession of all the essential attributes or all the parts necessary to the integrity of a substance. This is absolute, where all defect is precluded, such as the \textit{perfection} of God; or according to its kind, as in created things.
			\item[3] Moral \textit{perfection} is the complete possession of all moral excellence, as in the Supreme Being; or the possession of such moral qualities and virtues as a thing is capable of.
			\item[4] A quality, endowment or acquirement completely excellent, or of great worth. In this sense, the word has a plural. What tongue can her \textit{perfection}s tell?
			\item[5] An inherent or essential attribute of supreme or infinite excellence; or one perfect in its kind; as the \textit{perfection}s of God. The infinite power, holiness, justice, benevolence and wisdom of God are denominated his \textit{perfection}s.
			\item[6] Exactness; as, to imitate a model to \textit{perfection}.
		\end{compactitem}

				\begin{compactdesc}
					\item[BDAG 995b, PDF 542] \textbf{}
						\begin{compactitem}
							\item[1] \textbf{pertaining to meeting the highest standard}
							\item[1.a] of things, \textbf{\textit{perfect}}
							\item[1.a.$\alpha$] as acme of goodness, as adjective
							\item[1.a.$\beta$] as acme of badness
							\item[1.b] of persons who are fully up to standard in a certain respect and not satisfied with half-way measures \textbf{\textit{perfect, complete, expert}}
							\item[2] \textbf{pertaining to being mature, \textit{full-grown, mature, adult}}
							\item[2.a] adjective
							\item[2.b] substantive
							\item[3] \textbf{pertaining to being a cult initiate, \textit{initiated}}. As a technical term of the mystery religions, \gr{τέλειος} refers to one initiated into mystic rites
							\item[4] \textbf{pertaining to being fully developed in a moral sense}
							\item[4.a] of humans \textbf{\textit{perfect, fully developed}}
							\item[4.b] of God \textbf{\textit{perfect}}
						\end{compactitem}
					\item[CGL 1366a, PDF 1406] \textbf{}
						\begin{compactitem}
							\item[1] having reached one's appointed end; (of persons, animals) complete in growth, \textbf{full-grown, adult, mature}
							\item[2] (of persons) complete in quality, \textbf{accomplished, perfect}...\textbf{perfect in goodness} (NT)
							\item[3] (of sacrificial goats) \textbf{perfect, unblemished}
							\item[4] (generally, of objects, activities) complete in extent or quality; (of a piece of weaving) \textbf{complete, finished}...
							\item[5] (of values, relationships, abstract qualities) complete in every respect;
							\item[6] having reached fulfillment
							\item[7] relating to fulfillment of marriage
							\item[8] (of gods) bringing fulfillment (of their own purposes or the prayers of others)
							\item[9] (of an eagle, as a bird of augury) bringing fulfillment
						\end{compactitem}
					\item[LSJ 1769b, PDF 1840] \textbf{}
						\begin{compactitem}
							\item[I.1.a] \textit{perfect}, of victims, \textit{entire, without spot} or \textit{blemish}
							\item[I.1.b] in Dialects, \textit{fully constituted, valid}
							\item[I.2] of animals, \textit{full-grown}
							\item[I.3.a] of persons, \textit{accomplished, perfect in his kind}
							\item[I.3.b] of things
							\item[I.4] of prayers, vows, etc., \textit{fulfilled, accomplished}
							\item[I.5.a] of numbers, \textit{full, complete}
							\item[I.5.b] in arithmetic, of \textit{perfect} numbers, which are equal to the sum of their divisors
							\item[I.6] (special use about a bowl with Zeus?)
							\item[II] of the gods, \textit{having power to fulfil prayer, all-powerful}
							\item[III] = \gr{τελευταῖος}, \textit{last}
							\item[IV] (special use for a banquet)
							\item[V] (more special uses in various areas)
							\item[VI] name of a month at Epidaurus
							\item[VII] (uses of the adverb \gr{τελέως})
							\item[VIII] (comparative and superlative uses)
						\end{compactitem}
