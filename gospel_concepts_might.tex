\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{MIGHT-1828}
	
	% -----
	\subsubsection{Noun}
	% -----

		\begin{compactitem}
			\item[1] Strength; force; power; primarily and chiefly, bodily strength or physical power.
			\item[2] Political power or great achievements.
			\item[3] National strength; physical power or military force.
			\item[4] Valor with bodily strength; military prowess.
			\item[5] Ability; strength or application of means.
			\item[6] Strength or force of purpose.
			\item[7] Strength of affection.
			\item[8] Strength of light; splendor; effulgence.
		\end{compactitem}
		
	% -----
	\subsubsection{Auxiliary verb}
	% -----

		\begin{compactitem}
			\item \textit{noun preterit tense} of may. Had power or liberty. He \textit{might} go, or \textit{might} have gone.
			\item[1] It sometimes denotes was possible, implying ignorance of the fact in the speaker. Orders \textit{might} have been given for the purpose.
		\end{compactitem}
