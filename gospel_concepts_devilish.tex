\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{DEVILISH-1828}

	\begin{compactitem}
		\item[1] Partaking of the qualities of the devil; diabolical; very evil and mischievous; malicious; as a \textit{devilish} scheme; \textit{devilish} wickedness.
		\item[2] Having communication with the devil; pertaining to the devil.
		\item[3] Excessive; enormous; in a vulgar and ludicrous sense; as a \textit{devilish} cheat.
	\end{compactitem}
