\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{TRANSFIGURATION-1828}
	
	% -----
	\subsubsection{Transfiguration}
	% -----

		\begin{compactitem}
			\item[1] A change of form; particularly, the supernatural change in the personal appearance of our Savior on the mount.
			\item[2] A feast held by the Romish church on the 6th of August, in commemoration of the miraculous change above mentioned.
		\end{compactitem}
		
	% -----
	\subsubsection{Transfigure}
	% -----

		\begin{compactitem}
			\item[1] To transform; to change the outward form or appearance.
		\end{compactitem}
