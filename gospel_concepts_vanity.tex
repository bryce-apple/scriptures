	% -----
	\subsubsection{Vain (adjective)}
	% -----

		\begin{compactitem}
			\item[1] Empty; worthless; having no substance, value or importance.
			\item[2] Fruitless; ineffectual.
			\item[3] Proud of petty things, or of trifling attainments; elated with a high opinion of one's own accomplishments, or with things more showy than valuable; conceited.
			\item[4] Empty; unreal.
			\item[5] Showy; ostentatious.
			\item[6] Light; inconstant; worthless.
			\item[7] Empty; unsatisfying.
			\item[8] False; deceitful; not genuine; spurious.
			\item[9] Not effectual; having no efficacy.
		\end{compactitem}
