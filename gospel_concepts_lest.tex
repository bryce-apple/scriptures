\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{LEST-1828}

	\begin{compactitem}
		\item That not; for fear that.
		\item Ye shall not eat of it, neither shall ye touch it, lest ye die. \hyperref[GENESIS-3-3]{Genesis 3:3}.
			\begin{compactitem}
				\item The phrase may be thus explained. Ye shall not touch it; that separated or dismissed, ye die. That here refers to the preceding command or sentence; that being removed or not observed, the fact being not so, ye will die.
			\end{compactitem}
		\item Sin no more, lest a worse thing come to thee. \hyperref[JOHN-5-14]{John 5:14}.
			\begin{compactitem}
				\item Sin no more; that fact not taking place, a worse thing will happen to thee.
			\end{compactitem}
	\end{compactitem}
