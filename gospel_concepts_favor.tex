\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{FAVOR-1828}
	
	% -----
	\subsubsection{Noun}
	% -----

		\begin{compactitem}
			\item[1] Kind regard; kindness; countenance; propitious aspect; friendly disposition.
			\item[2] Support; defense; vindication; or disposition to aid, befriend, support, promote or justify.
			\item[3] A kind act or office; kindness done or granted; benevolence shown by word or deed; any act of grace or good will, as distinguished from acts of justice or renumeration.
			\item[4] Lenity; mildness or mitigation of punishment.
			\item[5] Leave; good will; a yielding or concession to another; pardon.
			\item[6] The object of kind regard; the person or thing favored.
			\item[7] A gift or present; something bestowed as an evidence of good will; a token of love; a knot of ribbons; something worn as a token of affection.
			\item[8] A feature; countenance. [Not used.]
			\item[9] Advantage; convenience afforded for success.
			\item[10] Partiality; bias.
		\end{compactitem}
	
	% -----
	\subsubsection{Verb}
	% -----

		\begin{compactitem}
			\item[1] To regard with kindness; to support; to aid or have the disposition to aid, or to wish success to; to be propitious to; to countenance; to befriend; to encourage. 
			\item[2] To afford advantages for success; to facilitate. 
			\item[3] To resemble in features.
			\item[4] To ease; to spare.
		\end{compactitem}
