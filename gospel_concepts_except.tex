	\label{EXCEPT-1828}
	
	% -----
	\subsubsection{Passive participle}
	% -----
		\label{EXCEPT-1828-participle}

		\begin{compactitem}
			\item Contracted from excepted. Taken out; not included. All were involved in this affair, \textit{except} one; that is, one excepted, the case absolute or independent clause. \textit{except} ye repent, ye shall all likewise perish; that is, \textit{except} this fact, that ye repent, or this fact being excepted, removed, taken away, ye shall all likewise perish. Or \textit{except} may be considered as the imperative mode. \textit{except} thou or ye, this fact, ye shall all likewise perish. Hence \textit{except} is equivalent to without, unless, and denotes exclusion.
		\end{compactitem}
