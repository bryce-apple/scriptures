\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{FULNESS-1828}

	\begin{compactitem}
		\item[1] The state of being filled, so as to leave no part vacant.
		\item[2] The state of abounding or being in great plenty; abundance.
		\item[3] Completeness; the state of a thing in which nothing is wanted; perfection.
		\item[4] Repletion; satiety; as from intemperance.
		\item[5] Repletion of vessels; as \textit{fullness} of blood.
		\item[6] Plenty; wealth; affluence.
		\item[7] Struggling perturbation; swelling; as the \textit{fullness} of the heart.
		\item[8] Largeness; extent.
		\item[9] Loudness; force of sound, such as fills the ear.
	\end{compactitem}
