			%
			\subparagraph{\texorpdfstring{\gr{λογίζομαι}}{}}
			%
				\label{LOGIZOMAI}
			
				\begin{compactdesc}
					\item[BDAG 529b] primarily a mathematical and accounting term, then of cognitive processes
						\begin{compactitem}
							\item[1] \textbf{to determine by mathematical process, \textit{reckon, calculate}}, frequently in a transferred sense
							\item[1.a] \textit{count, take into account} \gr{τι} \textit{something}
							\item[1.b] as a result of a calculation \textit{evaluate, estimate, look upon as, consider}
							\item[2] \textbf{to give careful thought to a matter, \textit{think (about), consider, ponder, let one's mind dwell on}} \gr{τι} \textit{something}
							\item[3] \textbf{to hold a view about something, \textit{think, believe, be of the opinion}}
						\end{compactitem}
					\item[CGL 877a, PDF 917] \textbf{}
						\begin{compactitem}
							\item[1] make an aritmetical calculation, \textbf{calculate, count}
							\item[2] (transitive) \textbf{count} ---\textit{items}...\textbf{calculate} ---\textit{interest}
							\item[3] reckon up the finances (of oneself or another); \textbf{do one's accounts}; (of officials) \textbf{audit accounts}
							\item[4] enter (a sum) in one's accounts (as a charge to another); \textbf{debit, charge} ---\textit{a sum}....\textbf{reckon up an account}...\textbf{claim in one's account}
							\item[5] employ rational calculation, \textbf{think, reflect, reason}...\textbf{make a plan}
							\item[6] (transitive) \textbf{take into account, bear in mind, consider}
							\item[7] \textbf{reckon, consider}
							\item[8] \textbf{count on, expect}...\textbf{reckon} (with future infinitive) \textit{that one will do something}
							\item[9] (passive) (of an argument) be reasoned through or formulated
						\end{compactitem}
					\item[LSJ 1055b, PDF 1126] \textbf{}
						\begin{compactitem}
							\item[I.1] properly of numerical calculation, \textit{count, reckon}
							\item[I.2] with accusative and infinitive, \textit{reckon} or \textit{calculate that}
							\item[I.3.a] \textit{set down to one's account}
							\item[I.3.b] \textit{audit the accounts} of a person
							\item[II.1] without reference to numbers, \textit{take into account, calculate, consider}
							\item[II.2] with accusative and infinitive, \textit{reckon, consider that}
							\item[II.3] with infinitive also, \textit{count} or \textit{reckon upon} doing, \textit{calculate} or \textit{expect} that
							\item[II.4] \textit{count upon}
							\item[II.5] \textit{conclude by reasoning, infer that}
							\item[II.6] absolute (examples of meaning given)
							\item[III] passive, mostly aorist and less frequently perfect
								\begin{compactitem}
									\item No other meanings are given under heading \#III, but it does list the uses in Romans 4.
								\end{compactitem}
						\end{compactitem}
				\end{compactdesc}
