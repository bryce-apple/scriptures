\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{DOOM-1828}

	\begin{compactitem}
		\item[1] Judgment; judicial sentence.
		\item[2] Condemnation; sentence; decree; determination affecting the fate or future state of another; usually a determination to inflict evil, sometimes otherwise.
		\item[3] That state to which one is doomed, or destined. To suffer misery is the \textit{doom} of sinners. To toil for subsistence is the \textit{doom} of most men.
		\item[4] Ruin; destruction.
		\item[5] Discrimination. [Not used.]
	\end{compactitem}
