				\begin{compactdesc}
					\item[BDAG 232b] \textbf{}
						\begin{compactitem}
							\item generally `capability', with emphasis on function
							\item[1] \textbf{potential for functioning in some way, \textit{power, might, strength, force, capability}}
							\item[1.a] generally
							\item[1.b] specifically, the \textit{power} that works wonders
							\item[2] \textbf{ability to carry out something, \textit{ability, capability}}
							\item[3] \textbf{a deed that exhibits ability to function powerfully, \textit{deed of power, miracle, wonder}}
							\item[4] \textbf{something that serves as an adjunct of power, \textit{resource}}
							\item[5] \textbf{an entity or being, whether human or transcendent, that functions in a remarkable manner, \textit{power}} as a personal transcendent spirit or heavenly agent/angel
							\item[6] \textbf{the capacity to convey thought, meaning}
						\end{compactitem}
					\item[CGL 397b, PDF 423] \textbf{}
						\begin{compactitem}
							\item[1] physical strength, \textbf{power, strength} (of persons or gods)
							\item[2] power or ability (to do something), \textbf{ability, capacity}
							\item[3] outward power (of persons or things), \textbf{influence, authority}
							\item[4] (specifically) \textbf{power} (of rules, nations, politicians)
							\item[5] military power, \textbf{forces}
							\item[6] \textbf{resources} (of a country); \textbf{quantity, abundance} (of wealth possessed by a people)
							\item[7] \textbf{natural property} or \textbf{capacity} (of a river, plants, the earth, mines); \textbf{effect} (of an embalming fluid); (generally) \textbf{practical effect} (of action, opposite theoretical importance)
							\item[8] capacity (inherentin or exercised by a person); \textbf{faculty} (of sight, or belonging to some part of the body)
							\item[9] \textbf{meaning} (of a word); \textbf{phonetic value, sound} (of a letter of the alphabet)
							\item[10] financial value, \textbf{worth, value} (of things)
							\item[11] (philosophy) \textbf{potentiality} (opposite actuality or activity)
							\item[12] (mathematics) number multiplied by itself (twice or three times), \textbf{power}; (specifically) \textbf{square number} or \textbf{square}
							\item[13] \textbf{square root} (especially as an irrational number)
						\end{compactitem}
					\item[LSJ 452a, PDF 523] \textbf{}
						\begin{compactitem}
							\item[I.1] \textit{power, might}, in Homer, especially of \textit{bodily strength}...generally, \textit{strength, power, ability} to do anything
							\item[I.2] \textit{outward power, influence, authority}
							\item[I.3] \textit{force for war, forces}
							\item[I.4] \textit{a power, quantity}
							\item[I.5] \textit{means}
							\item[II.1] \textit{power, faculty, capacity}...absolute, any \textit{natural capacity} or \textit{faculty}, that may be improved and may be used for good or ill
							\item[II.2.a] \textit{elementary force}, such as heat, cold, etc.
							\item[II.2.b] \textit{property, quality}
							\item[II.2.c] in plural, \textit{agencies}
							\item[II.2.d] \textit{function, meaning}, of part in whole
							\item[II.2.e] in Music, \textit{function, value}, of a note in the scale
							\item[II.3] \textit{faculty, art}, or \textit{craft}
							\item[II.4.a] \textit{a medicine}
							\item[II.4.b] \textit{action} of medicines
							\item[II.5] \textit{magically potent substance} or \textit{object}
							\item[III.1.a] \textit{force} or \textit{meaning} of a word
							\item[III.1.b] phonetic \textit{value} of sounds or letters
							\item[III.2] \textit{worth} or \textit{value} of money
							\item[IV] \textit{capability of existing} or \textit{acting, potentiality}, opposite \textit{actuality}
							\item[V.1.a] Mathematics, \textit{power}...usually \textit{second power, square}
							\item[V.1.b] \textit{square number} 
							\item[V.1.c] \textit{square} of an unknown quantity
							\item[V.2] \textit{square root} of a number which is not a perfect square
							\item[V.3] \textit{product} of two numbers
							\item[VI] concrete, \textit{powers}, especially of divine beings
							\item[VII] \textit{manifestation of divine power, miracle}
