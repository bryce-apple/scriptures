% Chip without mixing: takes explicit BACK and FRAME color names
\newcommand{\TagChipBF}[3]{%
  \begingroup
  \setlength{\fboxsep}{.5mm}%
  \fcolorbox{#2}{#1}{{\timelinefontsize\textsf{*#3}}}%
  \endgroup
}

% Entry macro: #1 date, #2 title, [#3 tags], +#4 body
\newcommand{\timelinefontsize}{\scriptsize} % or \footnotesize for smaller
% #1 = date (YYYY-MM-DD)
% #2 = title
% [#3] = tags (comma-separated)
% [#4] = location (optional)
% +#5 = body text
\NewDocumentCommand{\TimelineEntry}{ m m O{} O{} O{} +m }{%
  % Arguments:
  % #1 = date (YYYY-MM-DD)
  % #2 = title
  % [#3] = tags (optional)
  % [#4] = location (optional)
  % [#5] = sources (optional)
  % +#6 = body text
  \begin{tcolorbox}[timeline]
    {\timelinefontsize
    % --- Date and title ---
    \textbf{\DTMdate{#1}}\hspace{0.3em}\textbf{#2}\par

    % --- Optional location ---
    \ifstrempty{#4}{}{%
      \vspace{0.3mm}\textit{#4}\par
    }

    % --- Optional tags ---
    \ifstrempty{#3}{}{%
      \vspace{0.5mm}\Tags{#3}\par
    }

    % --- Body text ---
    #6

    % --- Optional sources ---
    \ifstrempty{#5}{}{%
      \vspace{0.8mm}%
      {\textsc{\textbf{Sources:}} #5\par}%
    }
    }
  \end{tcolorbox}
}
