				\begin{compactdesc}
					\item[BDAG 582a] \textbf{}
						\begin{compactitem}
							\item[1] \textit{commit adultery}
							\item[1a] of both sexes, with reference to the Ten Commandments
							\item[1b] of the man only
							\item[1b$\alpha$] absolute
							\item[1b$\beta$] with object \gr{τινα} (\gr{γυναῖκα}) \textit{commit adultery with someone}
						\end{compactitem}
					\item[LSJ 1141b, PDF 1212] \textbf{}
						\begin{compactitem}
							\item[I.1] \textit{commit adultery with} a woman, \textit{debauch} her, with accusative
							\item[I.2] metaphorical, \textit{worship idolatrously}
							\item[II] intransitive, \textit{commit adultery}
							\item[III] \textit{(A special use of the future middle of this verb.)}
						\end{compactitem}
				\end{compactdesc}
