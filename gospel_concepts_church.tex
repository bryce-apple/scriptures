\subsection{Key Phrases} % *KEYPHRASES *PHRASES
% ----------------------
	\label{CHURCH-PHRASES}
	
	% % NOTE THAT THIS SECTION IS **ONLY** FOR SCRIPTURES! USES BY JS, BY, OR OTHERS SHOULD GO BELOW IN THE ``RESTORATION USAGE'' SECTION!

	\begin{compactdesc}
	
		\item[church of Enoch] \textbf{1} | \hyperref[DC76-67]{DC 76:67}
		
		\item[church of the firstborn] \textbf{10} | \hyperref[COLOSSIANS-1-18]{Colossians 1:18}; \hyperref[HEBREWS-12-23]{Hebrews 12:23}; \hyperref[DC76-54]{DC 76:54}, \hyperref[DC76-67]{67}, \hyperref[DC76-71]{71}, \hyperref[DC76-94]{94}, \hyperref[DC76-102]{102}; \hyperref[DC77-11]{DC 77:11}, \hyperref[DC78-21]{DC 78:21}; \hyperref[DC88-5]{DC 88:5}; \hyperref[DC93-22]{DC 93:22}; \hyperref[DC107-19]{DC 107:19}
			\begin{compactdesc}
				\item[Bible anchor verses] \phantom{.}
					\begin{compactdesc}
						\item[Hebrews 12:23] \gr{καὶ ἐκκλησίᾳ πρωτοτόκων ἀπογεγραμμένων ἐν οὐρανοῖς, καὶ κριτῇ θεῷ πάντων, καὶ πνεύμασι δικαίων τετελειωμένων,}
					\end{compactdesc}
				\item[Commentary] Note that the ``firstborn'' bit in the Hebrews verse is in genitive \textit{plural}, not singular.
				
					\hyperref[DC76-94]{DC 76:94} says, ``They who dwell in his [God the Father's] presence are the church of the Firstborn.'' 
					
					\hyperref[DC93-22]{DC 93:22} says, ``And all those who are begotten through me are partakers of the glory of the same, and are the church of the Firstborn.''
			\end{compactdesc}
			
	\end{compactdesc}
