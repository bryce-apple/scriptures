	\label{ORDINATION-1828}
	
	% -----
	\subsubsection{Ordain}
	% -----

		\begin{compactitem}
			\item[1] Properly, to set; to establish in a particular office or order; hence, to invest with a ministerial function or sacerdotal power; to introduce and establish or settle in the pastoral office with the customary forms and solemnities; as, to \textit{ordain} a minister of the gospel. In America, men are ordained over a particular church and congregation, or as evangelists without the charge of a particular church, or as deacons in the episcopal church.
			\item[2] To appoint; to decree.
			\item[3] To set; to establish; to institute; to constitute.
			\item[4] To set apart for an office; to appoint.
			\item[5] To appoint; to prepare.
		\end{compactitem}
