	% -----
	\subsubsection{Transitive verb}
	% -----

		\begin{compactitem}
			\item To hoard; to collect and reposit, either money or other things, for future use; to lay up; as, to \textit{treasure} gold and silver; usually with \textit{up}.
		\end{compactitem}
