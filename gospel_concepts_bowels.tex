\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{BOWELS-1828}

	\begin{compactitem}
		\item[1] The intestines of an animal; the entrails, especially of man. The heart.
		\item[2] The interior part of any thing; as the bowels of the earth.
		\item[3] The seat of pity or kindness; hence, tenderness, compassion, a scriptural sense.
	\end{compactitem}

				\begin{compactdesc}
					\item[BDAG 834a]
						\begin{compactitem}
							\item[1] the inward parts of a body, including especially the viscera, \textit{inward parts, entrails}
							\item[2] as often in the ancient world, inner body parts served as referents for psychological aspects (see \gr{καρδία}): of the seat of the emotions, in our usage a transference is made to the rendering \textit{heart}...(on English ``bowels'' in this sense see OED s.v. bowel sb. 3), in our literature mostly as the seat and source of love...sympathy, and mercy
							\item[3] by metonymy [from meaning \#2], of the feeling itself, plural \textit{love, affection}
						\end{compactitem}
					\item[CGL 1280a, PDF 1320] (This word is listed as the middle/passive form, so the definitions below correspond to that form.)
						\begin{compactitem}
							\item[1] plural internal organs of a sacrificial animal; innards, offal (eaten by participants at a sacrifice)
							\item[2] (collective singular referring to human organs) \textbf{lungs} or \textbf{guts}; \textbf{spleen}
							\item[3] \textbf{womb}
							\item[4] (figurative, singular and plural) inner organs as the seat of emotions, \textbf{heart, guts} (as the seat of anxiety); (of impetuousness or violence); (of anger); (of pity) [NT mentioned here]; (of love); (of eager enthusiasm); (referring to one's innermost nature or character) \textbf{heart}
						\end{compactitem}
