\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{GOODLY-1828}x

	\begin{compactitem}
		\item Being of a handsome form; beautiful; graceful; as a \textit{goodly} person; \textit{goodly} raiment; \textit{goodly} houses.
		\item Pleasant; agreeable; desirable; as \textit{goodly} days.
		\item Bulky; swelling; affectedly turgid.
	\end{compactitem}

\subsection{Oxford English Dictionary} % *OED
% ----------------------
	\label{GOODLY-OED}

	\begin{compactitem}
		\item[1a] Of good or pleasing appearance; handsome, beautiful, good-looking; comely, fair.
		\item[1b] Of speech, words, etc.: pleasing, delightful.
		\item[2] Of good, noble, or valiant character; virtuous, honourable; (also) of distinguished attainments or bearing. \textit{(It tags this meaning as ``obsolete'' and the last instance cited comes from around 1600.)}
		\item[3] Gracious, courteous; friendly; gentle, kind. \textit{(Last instance cited here is around 1540.)}
		\item[4a] Of a good quality or high standard; splendid, excellent, fine. \textit{(Instances cited through the 1800's and one in 1994.)}
		\item[4b] In \textit{ironic} use.
		\item[5] Notable or considerable in respect of size, quantity, or number; fairly large, sizeable.
		\item[6] Convenient, proper; due, appropriate.
	\end{compactitem}
