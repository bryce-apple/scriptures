\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{ANSWER-1828}

	\begin{compactitem}
		\item[1] To speak in return to a call or question, or to a speech, declaration or argument of another person.
		\item[2] To be equivalent to; to be adequate to, or sufficient to accomplish the object.
		\item[3] To comply with, fulfill, pay or satisfy.
		\item[4] To act in return, or opposition.
		\item[5] To bear a due proportion to; to be equal or adequate; to suit.
		\item[6] To perform what was intended; to accomplish; as, the measure does not \textit{answer} its end; it does not \textit{answer} the purpose.
		\item[7] To be opposite to; to face.
		\item[8] To write in reply; to reply to another writing, by way of explanation, refutation or justification.
		\item[9] To solve, as a proposition or problem in mathematics.
	\end{compactitem}
