\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{PEACEABLE-1828}

	\begin{compactitem}
		\item Free from war, tumult or public commotion.
		\item[1] Free from private feuds or quarrels. 
		\item[2] Quiet; undisturbed; not agitated with passion.
		\item[3] Not violent, bloody or unnatural; as, to die a \textit{peaceable} death.
	\end{compactitem}

% ----------------------
\subsection{Oxford English Dictionary} % *OED
% ----------------------
	\label{PEACEABLE-OED}

	\begin{compactitem}
		\item[1.a] Of a person, animal, etc.: inclined to or in favour of peace; avoiding or inclined to avoid strife, violence, etc.; of a peaceful character; not aggressive or quarrelsome. Also, of an action, intention, etc.: not involving or tending towards aggression, argument, strife, violence, etc.
		\item[1.b] Not noisy or talkative; not restless; restrained or disciplined in behaviour. \textit{Obsolete.}
		\item[2.a] Characterized by peace; free from disturbance; quiet, tranquil.
		\item[2.b] Of the sea, weather, etc.: calm, peaceful; not stormy. \textit{Obsolete} (\textit{archaic} in later use).
	\end{compactitem}
