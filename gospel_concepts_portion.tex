\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{PORTION-1828}
	
	% -----
	\subsubsection{Noun}
	% -----

		\begin{compactitem}
			\item[1] In general, a part of any thing separated from it. Hence,
			\item[2] A part, though not actually divided, but considered by itself.
			\item[3] A part assigned; an allotment; a dividend.
			\item[4] The part of an estate given to a child or heir, or descending to him by law, and distributed to him in the settlement of the estate.
			\item[5] A wife's fortune.
		\end{compactitem}
		
	% -----
	\subsubsection{Verb}
	% -----
	
		\begin{compactitem}
			\item To divide; to parcel; to allot a share or shares.
			\item[1] To endow.
		\end{compactitem}
