			\subparagraph{\texorpdfstring{\he{mA+siy/a.h}}{}}

				\begin{compactdesc}
					\item[HALOT 5800, PDF 1622] \textbf{}
						\begin{compactitem}
							\item[1] the king of Israel, Saul, David and his descendants
							\item[2] Cyrus
							\item[3] priest, the anointed priest, the high priest
							\item[4] the patriarchs
							\item[5] ``Messiah'' (Hellenism from \he{m:+siy.hA'}), but not an eschatological saviour in OT (1 Samuel 2:10, Psalms 2:2, Daniel 9:25ff), first in Judaism, \gr{Χριστός} in NT
						\end{compactitem}
					\item[BDB 4899, PDF 1457] \textbf{}
						\begin{compactitem}
							\item anointed
							\item[1] king of Israel \textit{anointed} by divine command
							\item[2] high priest of Israel
							\item[3] Cyrus as commissioned by Yahweh
							\item[4] \textit{Messianic prince}
							\item[5] patriarchs, regarded as anointed kings
						\end{compactitem}
					\item[DCH PDF 2721] \textbf{}
						\begin{compactitem}
							\item anointed one
							\item[1] in reference to king
							\item[2] in reference to priest
							\item[3] in reference to prophets
							\item[4] in reference to eschatological figure, a ``messiah,'' whether royal or priestly \textit{(Comparing this meaning and HALOT \#5, I'm not sure whether the passages cited in DCH include the sense of a ``saviour,'' or just an end-of-days figure who is coming. But if such a figure had no power to save, or to at least do something, I'm not sure why you'd be very interested in that person.)}
						\end{compactitem}
				\end{compactdesc}
