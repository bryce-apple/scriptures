\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{PROSPER-1828}
	
	% -----
	\subsubsection{Transitive verb}
	% -----

		\begin{compactitem}
			\item To favor; to render successful. All things concur to \textit{prosper} our design.
		\end{compactitem}
		
	% -----
	\subsubsection{Intransitive verb}
	% -----

		\begin{compactitem}
			\item To be successful; to succeed.
			\item[1] To grow or increase; to thrive; to make gain; as, to \textit{prosper} in business. Our agriculture, commerce and manufactures now \textit{prosper}.
		\end{compactitem}
