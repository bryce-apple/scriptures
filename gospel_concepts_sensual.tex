\subsection{1828 American Dictionary of the English Language} % *1828
% ----------------------
	\label{SENSUAL-1828}

	\begin{compactitem}
		\item[1] Pertaining to the senses, as distinct from the mind or soul.
		\item[2] Consisting in sense, or depending on it; as \textit{sensual} appetites, hunger, lust, etc.
		\item[3] Affecting the senses, or derived from them; as \textit{sensual} pleasure or gratification. Hence,
		\item[4] In \textit{theology}, carnal; pertaining to the flesh or body, in opposition to the spirit; not spiritual or holy; evil.
		\item[5] Devoted to the gratification of sense; given to the indulgence of the appetites; sewd; luxurious.
	\end{compactitem}
